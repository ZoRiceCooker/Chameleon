"Abstract rule type"
abstract type MorphRule end

"Rule: when CPU > threshold, run an action"
struct CpuThresholdRule <: MorphRule
    threshold::Float64
    action::Function
end

function apply_rule(rule::CpuThresholdRule, snap::TelemetrySnapshot)
    if snap.cpu_usage > rule.threshold
        rule.action(snap)
    end
end

function apply_rules(rules::Vector{MorphRule}, snap::TelemetrySnapshot)
    for r in rules
        if r isa CpuThresholdRule
            apply_rule(r::CpuThresholdRule, snap)
        end
        # extend with more rule types
    end
end

high_cpu_rule = ChameleonOS.CpuThresholdRule(80.0) do snap
    println("High CPU: $(snap.cpu_usage) – surfacing diagnostic tools")
    ChameleonOS.pin_app_to_start("C:\\Windows\\System32\\Taskmgr.exe")
    ChameleonOS.launch_app("C:\\Windows\\System32\\Taskmgr.exe")
end

config = ChameleonOS.ChameleonConfig(
    2000,  # poll every 2 seconds
    [high_cpu_rule]
)

ChameleonOS.start_chameleon(config)
